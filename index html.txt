<script>
    // ⚠️ REEMPLAZA ESTA URL CON LA QUE COPIASTE DE APPS SCRIPT (la que termina en /exec)
    const API_URL = "https://script.google.com/macros/s/AKfycbyRDI7RJ2pneGNeac9CsYAscH1graj4mojw-LdHlthtAp910yYJy1gISzsgIu0wZHC0cQ/exec";

    let idFilaSeleccionada = null;

    function buscarMiembro() {
      const texto = document.getElementById("input-nombre").value.trim();
      const contenedor = document.getElementById("resultados-busqueda");
      if (!texto) return alert("Por favor escribe un nombre o apellido.");

      contenedor.innerHTML = "<p>Cargando coincidencias...</p>";

      // Cambiamos google.script.run por fetch()
      fetch(`${API_URL}?accion=buscar&texto=${encodeURIComponent(texto)}`)
        .then(res => res.json())
        .then(json => {
          contenedor.innerHTML = "";
          if (json.exito && json.resultados.length > 0) {
            json.resultados.forEach(item => {
              const div = document.createElement("div");
              div.className = "item-lista";
              div.innerText = item.nombreCompleto;
              div.onclick = () => seleccionarMiembro(item.idFila, item.nombreCompleto);
              contenedor.appendChild(div);
            });
          } else {
            contenedor.innerHTML = "<p>No se encontraron coincidencias.</p>";
          }
        })
        .catch(err => alert("Error al buscar: " + err));
    }

    function seleccionarMiembro(idFila, nombreCompleto) {
      idFilaSeleccionada = idFila;
      document.getElementById("nombre-seleccionado").innerText = nombreCompleto;
      document.getElementById("pantalla-busqueda").classList.add("oculto");
      document.getElementById("pantalla-validacion").classList.remove("oculto");
    }

    function volverABusqueda() {
      document.getElementById("pantalla-validacion").classList.add("oculto");
      document.getElementById("pantalla-busqueda").classList.remove("oculto");
    }

    function validarFecha() {
      const fecha = document.getElementById("input-fecha").value;
      if (!fecha) return alert("Ingresa tu fecha de nacimiento.");

      // Cambiamos google.script.run por fetch()
      fetch(`${API_URL}?accion=validar&idFila=${idFilaSeleccionada}&fecha=${encodeURIComponent(fecha)}`)
        .then(res => res.json())
        .then(json => {
          if (json.exito) {
            mostrarSubpaginaEdicion(json);
          } else {
            alert(json.mensaje);
          }
        })
        .catch(err => alert("Error al validar: " + err));
    }

    function mostrarSubpaginaEdicion(json) {
      document.getElementById("pantalla-busqueda").classList.add("oculto");
      document.getElementById("pantalla-validacion").classList.add("oculto");
      document.getElementById("pantalla-edicion").classList.remove("oculto");

      const gIdentidad = document.getElementById("grupo-identidad");
      const gContacto = document.getElementById("grupo-contacto");
      const gOtros = document.getElementById("grupo-otros");

      gIdentidad.innerHTML = "";
      gContacto.innerHTML = "";
      gOtros.innerHTML = "";

      let listaCampos = [];
      if (json.columnas && Array.isArray(json.columnas)) {
        listaCampos = json.columnas;
      } else if (json.datos && Array.isArray(json.datos)) {
        listaCampos = json.datos;
      } else {
        const fuenteDatos = json.datos || json;
        listaCampos = Object.keys(fuenteDatos).map(k => ({ nombre: k, valor: fuenteDatos[k] }));
      }

      const limpiarTexto = (str) => 
        String(str || "").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim();

      const ordenIdentidad = ["dni", "nombre", "apellido", "fecha", "llamemos"];
      const ordenContacto = ["correo", "whatsapp", "distrito", "tecnica", "san marcos", "grado", "honorario", "comision"];
      const camposProcesados = new Set();

      const opcionesDesplegables = {
        "cumpleanos": ["Un flyer que aparezca en el chat de miembros", "Un saludo rápido por chat/email.", "Prefiero que sea reservado, ¡pero gracias!"],
        "merito": ["Un flyer que aparezca en el chat de miembros", "Un saludo rápido por chat/email.", "Prefiero que sea reservado, ¡pero gracias!"],
        "base": ["20", "21", "22", "23", "24", "25", "26"],
        "grado": ["1er Año", "2do Año", "3er Año", "4to Año", "5to Año", "Egresado/a"],
        "comision": ["Académica", "Investigación", "RRPP", "Prensa", "Talento y Cultura"],
        "honorario": ["Sí", "No", "No sabría"]
      };

      function crearCampoHTML(keyOriginal, valorOriginal, esObligatorio = false) {
        const div = document.createElement("div");
        div.className = "campo-item";

        const label = document.createElement("label");
        const keyLimpia = limpiarTexto(keyOriginal);
        const esDNI = keyLimpia === "dni";

        label.innerHTML = esObligatorio || esDNI 
          ? `${keyOriginal} <span style="color:red;">* (Obligatorio)</span>` 
          : keyOriginal;

        const reglaDesplegable = Object.keys(opcionesDesplegables).find(patron => keyLimpia.includes(patron));
        let inputControl;

        if (reglaDesplegable) {
          inputControl = document.createElement("select");
          inputControl.name = keyOriginal;
          if (esObligatorio || esDNI) inputControl.required = true;
          inputControl.style.width = "100%";
          inputControl.style.padding = "10px";
          inputControl.style.marginTop = "5px";
          inputControl.style.border = "1px solid #cbd5e0";
          inputControl.style.borderRadius = "6px";

          const opcionDefecto = document.createElement("option");
          opcionDefecto.value = "";
          opcionDefecto.text = "-- Selecciona una opción --";
          inputControl.appendChild(opcionDefecto);

          opcionesDesplegables[reglaDesplegable].forEach(opcionTexto => {
            const opt = document.createElement("option");
            opt.value = opcionTexto;
            opt.text = opcionTexto;
            if (String(valorOriginal).trim().toLowerCase() === opcionTexto.toLowerCase()) {
              opt.selected = true;
            }
            inputControl.appendChild(opt);
          });
        } else {
          inputControl = document.createElement("input");
          inputControl.type = "text";
          inputControl.name = keyOriginal;
          inputControl.value = valorOriginal || "";
          if (esObligatorio || esDNI) inputControl.required = true;

          if (esDNI) {
            inputControl.className = "input-dni";
            inputControl.placeholder = "Ingresa tu DNI (mín. 7 dígitos)";
          }
        }

        div.appendChild(label);
        div.appendChild(inputControl);
        return div;
      }

      ordenIdentidad.forEach(patron => {
        listaCampos.forEach(item => {
          const keyLimpia = limpiarTexto(item.nombre);
          if (keyLimpia.includes(patron) && !camposProcesados.has(item.nombre)) {
            gIdentidad.appendChild(crearCampoHTML(item.nombre, item.valor, true));
            camposProcesados.add(item.nombre);
          }
        });
      });

      ordenContacto.forEach(patron => {
        listaCampos.forEach(item => {
          const keyLimpia = limpiarTexto(item.nombre);
          if (keyLimpia.includes(patron) && !camposProcesados.has(item.nombre)) {
            gContacto.appendChild(crearCampoHTML(item.nombre, item.valor, true));
            camposProcesados.add(item.nombre);
          }
        });
      });

      listaCampos.forEach(item => {
        if (!camposProcesados.has(item.nombre)) {
          gOtros.appendChild(crearCampoHTML(item.nombre, item.valor, false));
          camposProcesados.add(item.nombre);
        }
      });
    }

    function guardarCambios(e) {
      e.preventDefault();
      const form = document.getElementById("form-edicion");

      if (!form.checkValidity()) {
        form.reportValidity();
        return;
      }

      const formData = new FormData(form);
      let datosActualizados = {};
      let dniValor = "";

      const limpiarTexto = (str) => String(str || "").toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim();

      formData.forEach((value, key) => {
        datosActualizados[key] = value;
        if (limpiarTexto(key).includes("dni")) dniValor = String(value).trim();
      });

      const soloNumeros = /^\d+$/;
      if (!dniValor || dniValor.length <= 6 || !soloNumeros.test(dniValor)) {
        alert("El DNI es obligatorio y debe tener más de 6 dígitos numéricos.");
        return;
      }

      // Cambiamos google.script.run por fetch() POST
      fetch(API_URL, {
        method: "POST",
        body: JSON.stringify({
          idFila: idFilaSeleccionada,
          datos: datosActualizados
        })
      })
      .then(res => res.json())
      .then(json => {
        if (json.exito) {
          alert("¡Tus datos han sido actualizados exitosamente!");
          location.reload();
        } else {
          alert("Error: " + json.mensaje);
        }
      })
      .catch(err => alert("Error al guardar: " + err));
    }
</script>
